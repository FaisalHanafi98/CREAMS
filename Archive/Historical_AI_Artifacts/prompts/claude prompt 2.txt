namespace App\Http\Controllers;

use App\Models\Letter;
use App\Models\LetterTemplate;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Str;
use Barryvdh\DomPDF\Facade\Pdf;
use Illuminate\Support\Facades\Storage;

class LetterController extends Controller
{
    public function generate(Request $request)
    {
        try {
            Log::info('Letter generation initiated', [
                'user_id' => session('id'),
                'role' => session('role')
            ]);

            $validated = $request->validate([
                'template_id' => 'required|exists:letter_templates,id',
                'recipient_name' => 'required|string|max:255',
                'recipient_address' => 'required|string',
                'subject' => 'required|string|max:255',
                'content' => 'required|string',
                'letter_date' => 'required|date',
            ]);

            // Generate reference number
            $reference = 'LTR/' . date('Y/m/') . strtoupper(Str::random(6));

            // Create letter record
            $letter = Letter::create([
                'reference_number' => $reference,
                'template_id' => $validated['template_id'],
                'user_id' => session('id'),
                'recipient_name' => $validated['recipient_name'],
                'recipient_address' => $validated['recipient_address'],
                'subject' => $validated['subject'],
                'content' => $validated['content'],
                'letter_date' => $validated['letter_date'],
                'status' => 'generated'
            ]);

            Log::info('Letter record created', ['letter_id' => $letter->id]);

            // Generate PDF
            $pdfPath = $this->generatePDF($letter);
            
            // Update letter with PDF path
            $letter->update(['pdf_path' => $pdfPath]);

            Log::info('PDF generated successfully', [
                'letter_id' => $letter->id,
                'pdf_path' => $pdfPath
            ]);

            // Return download response
            return response()->download(storage_path('app/' . $pdfPath), 
                $reference . '.pdf')->deleteFileAfterSend(false);

        } catch (\Exception $e) {
            Log::error('Letter generation failed', [
                'error' => $e->getMessage(),
                'trace' => $e->getTraceAsString()
            ]);

            return redirect()->back()
                ->withErrors('Failed to generate letter. Please try again.')
                ->withInput();
        }
    }

    private function generatePDF(Letter $letter)
    {
        try {
            // Load template
            $template = LetterTemplate::findOrFail($letter->template_id);
            
            // Prepare data for PDF
            $data = [
                'letter' => $letter,
                'template' => $template,
                'header_image' => $template->header_image ? 
                    storage_path('app/public/' . $template->header_image) : null,
                'footer_image' => $template->footer_image ? 
                    storage_path('app/public/' . $template->footer_image) : null,
                'organization' => [
                    'name' => 'CREAMS Rehabilitation Centre',
                    'address' => 'IIUM Gombak, Selangor, Malaysia',
                    'phone' => '+603-6196-4000',
                    'email' => 'info@creams.edu.my'
                ]
            ];

            // Generate PDF
            $pdf = PDF::loadView('letters.pdf', $data);
            
            // Set paper size and orientation
            $pdf->setPaper('A4', 'portrait');
            
            // Save PDF
            $filename = 'letters/' . date('Y/m/') . $letter->reference_number . '.pdf';
            Storage::makeDirectory('letters/' . date('Y/m/'));
            $pdf->save(storage_path('app/' . $filename));
            
            return $filename;

        } catch (\Exception $e) {
            Log::error('PDF generation failed', [
                'error' => $e->getMessage(),
                'letter_id' => $letter->id
            ]);
            throw $e;
        }
    }

    public function download($id)
    {
        try {
            $letter = Letter::findOrFail($id);
            
            // Check permission
            if (session('role') !== 'admin' && $letter->user_id !== session('id')) {
                abort(403, 'Unauthorized access');
            }

            if (!$letter->pdf_path || !Storage::exists($letter->pdf_path)) {
                // Regenerate PDF if missing
                $letter->pdf_path = $this->generatePDF($letter);
                $letter->save();
            }

            return response()->download(
                storage_path('app/' . $letter->pdf_path),
                $letter->reference_number . '.pdf'
            );

        } catch (\Exception $e) {
            Log::error('Letter download failed', [
                'error' => $e->getMessage(),
                'letter_id' => $id
            ]);

            return redirect()->back()
                ->withErrors('Failed to download letter.');
        }
    }
}
